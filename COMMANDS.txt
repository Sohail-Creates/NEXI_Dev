# NEXI HTTPS service startup commands
# Run each section in a separate PowerShell terminal opened at the repository root.
# Stop any older instance using the same port before starting its replacement.

================================
CENTRAL
================================
cd 01_central_server
.\venv\Scripts\Activate.ps1
$repo = (Resolve-Path ..).Path
$env:PYTHONPATH = $repo
python -m uvicorn main:app --host 127.0.0.1 --port 8000 --env-file "$repo\.env" --ssl-certfile "$repo\config\certificates\nexi-local.crt" --ssl-keyfile "$repo\config\certificates\nexi-local.key"
