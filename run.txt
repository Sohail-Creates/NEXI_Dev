# NEXI native run guide -- Windows PowerShell, Python 3.11.
# Prerequisite: `python --version` must report Python 3.11 on PATH.
# Open SEVEN terminals, each at this repository root. Copy only the section
# for that terminal. Each service uses its own isolated venv; do not use the
# root monolithic requirements.txt. Ctrl+C stops that terminal's service.
# The venv-audit-handover environments are the fresh environments verified by
# the final audit. On a fresh clone, each section creates its own environment.
# First-time prerequisites (do not overwrite an existing .env or key):
#   Copy .env.example to .env ONLY if .env does not exist. Configure the shared
#   JWT/service secrets, Fernet key/file, and provider credentials securely.
#   AUTH_ENFORCEMENT_ENABLED=true; ENABLE_ENCRYPTION=true; TLS enabled; no HTTP.
#   Native service URLs use https://localhost:<port>, not Docker DNS names.
#   After installing any service's manifest, provision local test certificates
#   ONCE with: python -m config.ssl_config --generate-local
#   This uses default config/certificates paths. For custom paths, provision
#   matching cert/key/CA yourself. Never regenerate certs while services run.
#   Local self-signed certificates expire; commercial CA provisioning is an
#   operator responsibility. Clients verify the configured CA, never bypass it.
#   The start commands below validate configured TLS files before serving.
# If activation is blocked by your PowerShell policy, permit scripts for this
# terminal only: Set-ExecutionPolicy -Scope Process -ExecutionPolicy RemoteSigned
# Missing physical hardware, DeepFace, Porcupine access, Piper weights, or
# provider credit is reported honestly as degraded; HTTP 200 degraded is alive,
# not proof that every hardware/provider-dependent feature works.

# === Central -- port 8000 ===
if (-not (Test-Path .\01_central_server\venv-audit-handover\Scripts\python.exe)) { python -m venv .\01_central_server\venv-audit-handover }
& .\01_central_server\venv-audit-handover\Scripts\Activate.ps1
python -m pip install -r .\01_central_server\requirements.txt
python -m pip check
# First-time semantic model provisioning ONLY (network/cache preparation):
# python -c "from shared.semantic_embeddings import SEMANTIC_EMBEDDING_MODEL; from sentence_transformers import SentenceTransformer; SentenceTransformer(SEMANTIC_EMBEDDING_MODEL)"
# Runtime embeddings use the locally provisioned artifact, with no fallback.
# Initialize an empty store idempotently; this does not import legacy JSON.
# For legacy JSON, run migrate_sqlite.py with --source-dir before first startup.
python -c "import sys; from dotenv import load_dotenv; load_dotenv('.env'); sys.path.insert(0, '01_central_server'); from sqlite_store import initialize; initialize()"
python -c "import sys; from dotenv import load_dotenv; load_dotenv('.env'); sys.path.insert(0, '01_central_server'); from sqlite_store import DATABASE; from migrate_cloud_sync_outbox import migrate_outbox; migrate_outbox(DATABASE)"
$tls = python -c "import json; from dotenv import load_dotenv; load_dotenv('.env'); from config.ssl_config import get_tls_config; t=get_tls_config(); t.validate(); assert t.enabled and not t.plaintext_transition; print(json.dumps({'cert':str(t.cert_file),'key':str(t.key_file)}))" | ConvertFrom-Json
python -m uvicorn main:app --app-dir .\01_central_server --host 127.0.0.1 --port 8000 --env-file .env --ssl-certfile $tls.cert --ssl-keyfile $tls.key
# Health: https://localhost:8000/health
